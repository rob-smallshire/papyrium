# A map of the Star LC-10 Colour's program, firmware version 1.5, as traced
# so far: where it keeps data rather than code, and what its routines and
# RAM locations do, in our own words. tools/lc10-rom/disassemble.py reads
# it to lay out and annotate the disassembly; docs/star-lc10-firmware.md
# explains what it shows.
#
# Each line is a CPU address or range, a kind, a name and a description.
# Kinds: data (bytes), commands (a command table: a byte and its handler's
# address, three bytes an entry, ended by a zero), code (a routine or an
# entry point within one), ram (a zero-page or RAM location).

# -- Data in the program -----------------------------------------------------

$8000-$80FF  data      program_data       Settings and the international sets: ESC R 1 at $8040 up to 12 at $80C4, then USA at $80D0
$B175-$B1A5  commands  standard_controls  Standard mode's control codes
$B1A6-$B1D6  commands  ibm_controls       IBM mode's control codes
$B1F4-$B2DB  commands  standard_escapes   Standard mode's escape commands
$B2F9-$B3E0  commands  ibm_escapes        IBM mode's escape commands
$B3F7-$B3F7  commands  standard_fs        Standard mode's commands after FS: none
$B40E-$B423  commands  ibm_fs             IBM mode's commands after FS: Star's
$905B-$905F  data      fill_905b          Unused
$B139-$B166  data      self_test_text     The self-test's heading: "DIP-SW" and the switch numbers
$B452-$B46F  data      fill_b452          Unused
$E810-$E81F  data      nlq_typefaces      The NLQ index table of each typeface: pica, then elite at $E818

# -- The main loop and command dispatch ---------------------------------------

$8100  code  reset                     Power on
$83B8  code  next_byte                 Take the next input byte and act on it, unless deselected ($39 bit 4); handlers end here, and unknown escapes come here
$B1D7  code  standard_escape           ESC in Standard mode: read the command and look it up in standard_escapes
$B2DC  code  ibm_escape                ESC in IBM mode: read the command and look it up in ibm_escapes
$B3E1  code  standard_fs_command       FS in Standard mode: read a byte and look it up in standard_fs, which is empty
$B3F8  code  ibm_fs_command            FS in IBM mode: read a byte and look it up in ibm_fs
$8423  code  choose_control_table      A control code: look it up in ibm_controls in IBM mode ($37 bit 5), else standard_controls
$87C2  code  carriage_return           Print the line and return the head to the left margin
$8A6B  code  reset_tabs                Horizontal stops every 8 columns from 8, or from 9 in IBM mode; called by ESC @ and IBM mode's ESC R
$9060  code  read_switch_modes         At power on, set $3A and $3B from the DIP switches ($37, $38)
$94D3  code  request_nlq               NLQ asked for ($5C bit 4); it prints ($5C bit 2) unless emphasized, a script or condensed
$94FA  code  cancel_nlq                Draft: clear $5C bits 2 and 4
$959E  code  download_draft_mode       ESC I 4 and 5: the download set in draft, emphasized off; nothing with the RAM a buffer
$95B6  code  download_nlq_mode         ESC I 6: the download set in NLQ whatever the modes, emphasized and condensed off; nothing with the RAM a buffer
$9718  code  set_typeface              The NLQ typeface in Y, 0-3, into $A1 bits 4-5
$A757  code  ibm_line_number           In IBM mode, a vertical stop's line n less one, unless 1
$B424  code  find_command              Look the command in A up in the table at ($3F); carry set if found, $3F-$40 then the handler
$B44A  code  read_command_byte         Read the next byte of a command
$B44E  code  flush_line                Print what the line buffer holds, before a change of mode (calls $8733)
$B4D4  code  read_byte                 Read the next input byte
$B52E  code  read_flag                 Read an argument, turning the digits "0" and "1" into 0 and 1

# -- RAM ---------------------------------------------------------------------

$37    ram  dip_switches_1   DIP switches 1-1 to 1-8 as bits 0-7, a bit set for a switch off; bit 5 (1-6 off) is IBM mode, which ESC 0x01 and 0x02 also change
$38    ram  dip_switches_2   DIP switches 2-1 to 2-4 as bits 0-3, set for off
$39    ram  panel_state      Bit 0 the pitch lock, bit 1 the style lock, bit 4 deselected, bits 5-6 the input state
$3A    ram  modes            Bit 2 the IBM table at 128-255, bit 3 IBM character set #2, bit 6 LF returns the head (switch 1-2 on), bit 7 CR feeds too (switch 1-8 off, IBM ESC 5)
$3B    ram  memory_modes     Bit 4 the RAM is a buffer, not download characters (switch 2-1)
$3C    ram  dispatch_state   Bit 2: a handler returns to its caller rather than to next_byte (IBM ESC P)
$3F    ram  table_pointer    The command table being searched, then the handler found ($3F-$40)
$41    ram  pitch            Bit 0 pica, bit 1 elite, bit 2 condensed, bit 4 printing codes from the All Characters Chart (IBM ESC \ and ^)
$42    ram  print_attributes Bit 0 double width, bit 1 italic, bit 2 underline, bit 3 the download set, bit 5 proportional, bit 7 overline
$5C    ram  print_mode       Bit 1 double-strike, bit 2 NLQ printing, bit 3 ESC W's double width, bit 4 NLQ asked for, bit 5 subscript, bit 6 superscript, bit 7 emphasized
$5D    ram  character_size   One bit for each ESC h size
$A1    ram  nlq_typeface     Bits 4-5 the ESC k typeface
$0112  ram  ibm_spacing      IBM mode's ESC A spacing, in 1/432 inch steps, for ESC 2; 72 at ESC @
$0121  ram  chart_count      Bytes left to print from the All Characters Chart (IBM ESC \ and ^), to $0122
$0234  ram  tab_stops        Horizontal stops as column numbers, after the pitch they were set in
